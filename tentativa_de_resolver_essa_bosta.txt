void questao_02(void) {
    Utility_Init();
    DAC_Init(DAC_CHANNEL2); 
    const float freq_hz = 100.0f; 
    const float vpp     = 2.0f;   
    const float vm      = 1.65f;  
    const float pi      = 3.14f;

    uint16_t lut_seno[200];
    float vp = vpp / 2.0f;

    for (int i = 0; i < 200; i++) {
        float tensao = vm + vp * sinf((2.0f * pi * i) / 200.0f);
        int32_t codigo = (int32_t)roundf((tensao / 3.3f) * 4095.0f);

        if (codigo > 4095) codigo = 4095;
        if (codigo < 0)    codigo = 0;
        lut_seno[i] = (uint16_t)codigo;
    }

    uint32_t atraso_us = (uint32_t)(5000.0f / freq_hz); // 1.000.000 / (200 * freq_hz)

    while (1) {
        for (int i = 0; i < 200; i++) {
            DAC_SetValue(DAC_CHANNEL2, lut_seno[i], DAC_RES_12BITS);
            Delay_us(atraso_us);
        }
    }
}








void questao_03(void) {
    Utility_Init();
    USART1_Init();

    const float limite_inf_v = 1.00f;
    const float limite_sup_v = 2.50f;

    ADC_Init(ADC1, SINGLE_CHANNEL, ADC_RES_12BITS);
    ADC_SingleChannel(ADC1, ADC_IN0); 


    ADC1->LTR = (uint16_t)((limite_inf_v / 3.3f) * 4095.0f);
    ADC1->HTR = (uint16_t)((limite_sup_v / 3.3f) * 4095.0f);
    ADC1->CR1 |= ADC_CR1_AWDEN; 

    while (1) {
        uint16_t leitura = ADC_GetSingleConversion(ADC1);
        float tensao = ((float)leitura * 3.3f) / 4095.0f;

        printf("Tensao = %.2f V\n", tensao);

        if (ADC1->SR & ADC_SR_AWD) {
            printf("ALERTA: Tensao fora da faixa segura (%.2f V a %.2f V)!\n",
                   limite_inf_v, limite_sup_v);
            ADC1->SR = ~ADC_SR_AWD; // Limpa exclusivamente o bit AWD
        }

        Delay_ms(500);
    }
}

void questao_06(void) {
    Utility_Init();
    DAC_Init(DAC_CHANNEL1); 

    const uint32_t tamanho_audio = sizeof(audio_raw) / sizeof(audio_raw[0]);

    // Configura TIM6 para taxa exata de 44.100 Hz (84 MHz / 1905 = 44.094 Hz)
    RCC->APB1ENR |= RCC_APB1ENR_TIM6EN;
    TIM6->PSC = 0;
    TIM6->ARR = 1904;
    TIM6->CR1 |= TIM_CR1_CEN;

    while (1) {
        for (uint32_t i = 0; i < tamanho_audio; i++) {
            while (!(TIM6->SR & TIM_SR_UIF)); // Aguarda o período de 1/44100 s (~22,68 us)
            TIM6->SR &= ~TIM_SR_UIF;

            DAC_SetValue(DAC_CHANNEL1, audio_raw[i], DAC_RES_8BITS);
        }

        Delay_ms(1000); 
    }
}



void questao_09(void) {
    Utility_Init();
    USART1_Init();

    GPIO_Clock_Enable(GPIOA);
    GPIO_Pin_Mode(GPIOA, PIN_0, INPUT);
    GPIO_Resistor_Enable(GPIOA, PIN_0, PULL_DOWN);

    // TIM5 (32 bits): 84 MHz / 8400 = 10 kHz -> 0,1 ms por tick
    RCC->APB1ENR |= RCC_APB1ENR_TIM5EN;
    TIM5->PSC = 8399;
    TIM5->ARR = 0xFFFFFFFF;
    TIM5->EGR = TIM_EGR_UG;

    while (1) {
        while (GPIO_Read_Pin(GPIOA, PIN_0) == HIGH); // Aguarda nível LOW inicial
        while (GPIO_Read_Pin(GPIOA, PIN_0) == LOW);  // Aguarda borda de subida

        TIM5->CNT = 0;
        TIM5->CR1 |= TIM_CR1_CEN;

        uint8_t timeout_5s = 0;
        while (GPIO_Read_Pin(GPIOA, PIN_0) == HIGH) {
            if (TIM5->CNT > 50000) { // 50000 * 0,1 ms = 5000 ms (5 s)
                timeout_5s = 1;
                break;
            }
        }

        TIM5->CR1 &= ~TIM_CR1_CEN;

        if (timeout_5s) {
            printf("Erro: O pulso e demasiadamente longo para ser medido (> 5s).\n");
        } else {
            float largura_ms = (float)TIM5->CNT / 10.0f;
            printf("Largura do pulso = %.1f ms\n", largura_ms);
        }
    }
}
